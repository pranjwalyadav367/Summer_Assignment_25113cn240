//Write a program to Write function to find factorial.
#include <stdio.h>
long long factorial(int n) {
    long long fact = 1;
    for (int i = 1; i <= n; i++) {
        fact *= i;
    }
    return fact;
}

int main() {
    int num;
    printf("enter the num :");
    scanf("%d",&num);
    printf("Factorial of %d is %lld\n", num, factorial(num));
    return 0;
}
